CREATE DATABASE УправлениеРестораном;
GO

USE УправлениеРестораном;
GO

SET NOCOUNT ON;
GO

IF OBJECT_ID(N'dbo.ВьюПрибыльПоЗаказам', N'V') IS NOT NULL DROP VIEW dbo.ВьюПрибыльПоЗаказам;
GO

IF OBJECT_ID(N'dbo.ПозицииЗаказа', N'U') IS NOT NULL DROP TABLE dbo.ПозицииЗаказа;
IF OBJECT_ID(N'dbo.Заказы', N'U') IS NOT NULL DROP TABLE dbo.Заказы;
IF OBJECT_ID(N'dbo.Брони', N'U') IS NOT NULL DROP TABLE dbo.Брони;
IF OBJECT_ID(N'dbo.Блюда', N'U') IS NOT NULL DROP TABLE dbo.Блюда;
IF OBJECT_ID(N'dbo.Столики', N'U') IS NOT NULL DROP TABLE dbo.Столики;
IF OBJECT_ID(N'dbo.Клиенты', N'U') IS NOT NULL DROP TABLE dbo.Клиенты;
GO

CREATE TABLE dbo.Клиенты
(
    ИдКлиента INT IDENTITY(1,1) PRIMARY KEY,
    ФИО NVARCHAR(120) NOT NULL,
    Телефон NVARCHAR(30) NOT NULL,
    Заметка NVARCHAR(400) NULL,
    ДатаСоздания DATETIME2 NOT NULL DEFAULT SYSDATETIME()
);
GO

CREATE TABLE dbo.Столики
(
    ИдСтолика INT IDENTITY(1,1) PRIMARY KEY,
    НомерСтолика INT NOT NULL UNIQUE,
    Вместимость INT NOT NULL,
    Статус NVARCHAR(20) NOT NULL,
    ЗанятоМест INT NOT NULL DEFAULT 0,
    ВремяСтатуса DATETIME2 NULL,
    ВремяБрони DATETIME2 NULL,
    CONSTRAINT Ограничение_Столики_Вместимость CHECK (Вместимость > 0),
    CONSTRAINT Ограничение_Столики_Статус CHECK (Статус IN (N'Свободен', N'Занят', N'Забронирован')),
    CONSTRAINT Ограничение_Столики_ЗанятоМест CHECK (ЗанятоМест >= 0 AND ЗанятоМест <= Вместимость)
);
GO

CREATE TABLE dbo.Блюда
(
    ИдБлюда INT IDENTITY(1,1) PRIMARY KEY,
    Название NVARCHAR(150) NOT NULL UNIQUE,
    Категория NVARCHAR(30) NOT NULL,
    Цена DECIMAL(10,2) NOT NULL,
    Себестоимость DECIMAL(10,2) NOT NULL,
    Доступно BIT NOT NULL DEFAULT 1,
    ДатаСоздания DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT Ограничение_Блюда_Категория CHECK (Категория IN (N'Супы', N'Салаты', N'Горячее', N'Десерты', N'Напитки')),
    CONSTRAINT Ограничение_Блюда_Цена CHECK (Цена > 0),
    CONSTRAINT Ограничение_Блюда_Себестоимость CHECK (Себестоимость >= 0)
);
GO

CREATE TABLE dbo.Брони
(
    ИдБрони INT IDENTITY(1,1) PRIMARY KEY,
    ИдКлиента INT NOT NULL,
    ИдСтолика INT NULL,
    ДатаБрони DATE NOT NULL,
    ВремяБрони TIME NOT NULL,
    КоличествоГостей INT NOT NULL,
    Предпочтение NVARCHAR(200) NULL,
    Комментарий NVARCHAR(500) NULL,
    Статус NVARCHAR(20) NOT NULL,
    ДатаСоздания DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT ВнешнийКлюч_Брони_Клиенты FOREIGN KEY (ИдКлиента) REFERENCES dbo.Клиенты(ИдКлиента),
    CONSTRAINT ВнешнийКлюч_Брони_Столики FOREIGN KEY (ИдСтолика) REFERENCES dbo.Столики(ИдСтолика),
    CONSTRAINT Ограничение_Брони_КоличествоГостей CHECK (КоличествоГостей > 0),
    CONSTRAINT Ограничение_Брони_Статус CHECK (Статус IN (N'Новая', N'Подтверждена', N'Отменена'))
);
GO

CREATE TABLE dbo.Заказы
(
    ИдЗаказа INT IDENTITY(1,1) PRIMARY KEY,
    ИдСтолика INT NOT NULL,
    ИдБрони INT NULL,
    ИдКлиента INT NULL,
    Статус NVARCHAR(20) NOT NULL,
    Комментарий NVARCHAR(400) NULL,
    ДатаСоздания DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ДатаЗакрытия DATETIME2 NULL,
    CONSTRAINT ВнешнийКлюч_Заказы_Столики FOREIGN KEY (ИдСтолика) REFERENCES dbo.Столики(ИдСтолика),
    CONSTRAINT ВнешнийКлюч_Заказы_Брони FOREIGN KEY (ИдБрони) REFERENCES dbo.Брони(ИдБрони),
    CONSTRAINT ВнешнийКлюч_Заказы_Клиенты FOREIGN KEY (ИдКлиента) REFERENCES dbo.Клиенты(ИдКлиента),
    CONSTRAINT Ограничение_Заказы_Статус CHECK (Статус IN (N'Готовится', N'Готов', N'Завершён', N'Отменён'))
);
GO

CREATE TABLE dbo.ПозицииЗаказа
(
    ИдПозицииЗаказа INT IDENTITY(1,1) PRIMARY KEY,
    ИдЗаказа INT NOT NULL,
    ИдБлюда INT NOT NULL,
    Количество INT NOT NULL,
    ЦенаЗаЕдиницу DECIMAL(10,2) NOT NULL,
    СебестоимостьЗаЕдиницу DECIMAL(10,2) NOT NULL,
    CONSTRAINT ВнешнийКлюч_ПозицииЗаказа_Заказы FOREIGN KEY (ИдЗаказа) REFERENCES dbo.Заказы(ИдЗаказа) ON DELETE CASCADE,
    CONSTRAINT ВнешнийКлюч_ПозицииЗаказа_Блюда FOREIGN KEY (ИдБлюда) REFERENCES dbo.Блюда(ИдБлюда),
    CONSTRAINT Ограничение_ПозицииЗаказа_Количество CHECK (Количество > 0),
    CONSTRAINT Ограничение_ПозицииЗаказа_Цена CHECK (ЦенаЗаЕдиницу > 0),
    CONSTRAINT Ограничение_ПозицииЗаказа_Себестоимость CHECK (СебестоимостьЗаЕдиницу >= 0)
);
GO

CREATE INDEX Индекс_Брони_ДатаВремя ON dbo.Брони(ДатаБрони, ВремяБрони);
CREATE INDEX Индекс_Заказы_СтатусДата ON dbo.Заказы(Статус, ДатаСоздания DESC);
CREATE INDEX Индекс_ПозицииЗаказа_Заказ ON dbo.ПозицииЗаказа(ИдЗаказа);
GO

CREATE VIEW dbo.ВьюПрибыльПоЗаказам
AS
SELECT
    з.ИдЗаказа,
    с.НомерСтолика,
    з.Статус,
    з.ДатаСоздания,
    з.ДатаЗакрытия,
    SUM(п.Количество * п.ЦенаЗаЕдиницу) AS Выручка,
    SUM(п.Количество * п.СебестоимостьЗаЕдиницу) AS Себестоимость,
    SUM(п.Количество * п.ЦенаЗаЕдиницу) - SUM(п.Количество * п.СебестоимостьЗаЕдиницу) AS Прибыль
FROM dbo.Заказы з
INNER JOIN dbo.Столики с ON с.ИдСтолика = з.ИдСтолика
INNER JOIN dbo.ПозицииЗаказа п ON п.ИдЗаказа = з.ИдЗаказа
GROUP BY
    з.ИдЗаказа,
    с.НомерСтолика,
    з.Статус,
    з.ДатаСоздания,
    з.ДатаЗакрытия;
